   9  /// What an action that replaces the current plan does with the plan it
  10  /// leaves (S6.3; U1, ADR-019 §3).
  11  enum LeavingPlan {
  12    /// Nothing to discard: the plan has no unsaved changes, or was just saved.
  13    /// An untouched plan that was never saved is still deleted by the switch.
  14    keep,
  15  
  16    /// The user chose Discard: a never-saved plan is deleted; a saved plan
  17    /// with changes goes back to what was saved (S4-DEF-04 = R).
  18    discard,
  19  }
  20  
  21  /// Asks before New plan, Copy to another night or Open replaces a plan with
  22  /// unsaved changes (S6.3; U1; supersedes S1.6's guard): Save · Discard ·
  23  /// Cancel. Save saves the plan as the Save plan button does, then goes on; a
  24  /// failed Save stops and says so. Cancel, back and a tap outside stop, and
  25  /// nothing changes. Null means stop; otherwise the caller goes on and passes
  26  /// the answer to the lifecycle (`discard:`).
  27  Future<LeavingPlan?> askBeforeLeavingPlan(BuildContext context) async {
  28    final plan = context.read<SessionPlanViewModel>();
  29    if (!plan.hasUnsavedChanges) return LeavingPlan.keep;
  30    final target = plan.selectedTarget;
  31    final choice = await askUnsavedChanges(
  32      context,
  33      plan: target == null ? 'This plan' : target.commonName ?? target.catalogId,
  34    );
  35    switch (choice) {
  36      case UnsavedChoice.cancel:
  37        return null;
  38      case UnsavedChoice.discard:
  39        return LeavingPlan.discard;
  40      case UnsavedChoice.save:
  41        if (!context.mounted) return null;
  42        final saved = await runWithFeedback(
  43          context,
  44          'save the plan',
  45          context.read<CaptureAnalysisViewModel>().saveSession,
  46        );
  47        return saved ? LeavingPlan.keep : null;
  48    }
  49  }
