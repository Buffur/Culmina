 256    /// Adds the palette and the shared type scale (no text under 12 sp).
 257    /// The scale's sizes and weights override Material's geometry; the
 258    /// colours stay the theme's.
 259    static ThemeData _withTokens(
 260      ThemeData base,
 261      AppPalette palette, {
 262      Color? textColor,
 263      bool field = false,
 264    }) {
 265      final hybrid = field
 266          ? HybridTokens.field
 267          : base.brightness == Brightness.dark
 268          ? HybridTokens.dark
 269          : HybridTokens.light;
 270      TextTheme readable(TextTheme t) {
 271        final scaled = t.merge(AppTypography.scale);
 272        return textColor == null
 273            ? scaled
 274            : scaled.apply(bodyColor: textColor, displayColor: textColor);
 275      }
 276  
 277      return _withControls(
 278        base.copyWith(
 279          extensions: [palette, hybrid],
 280          pageTransitionsTheme: const PageTransitionsTheme(
 281            builders: {
 282              TargetPlatform.android: HybridPageTransitionsBuilder(),
 283              TargetPlatform.windows: HybridPageTransitionsBuilder(),
 284              TargetPlatform.linux: HybridPageTransitionsBuilder(),
 285              TargetPlatform.iOS: CupertinoPageTransitionsBuilder(),
 286              TargetPlatform.macOS: CupertinoPageTransitionsBuilder(),
 287            },
 288          ),
 289          textTheme: readable(base.textTheme),
 290          primaryTextTheme: readable(base.primaryTextTheme),
 291        ),
 292        palette,
 293        field: field,
 294      );
 295    }

 297    /// The controls' look (S5.2; `docs/DESIGN_SYSTEM.md`): the button roles,
 298    /// text fields, dialogs, sheets, menus and messages, from the tokens
 299    /// only, so field mode stays red or black.
 300    static ThemeData _withControls(
 301      ThemeData t,
 302      AppPalette p, {
 303      required bool field,
 304    }) {
 305      final s = t.colorScheme;
 306      final hybrid = t.extension<HybridTokens>()!;
 307      final text = t.textTheme;
 308      final shape = WidgetStatePropertyAll(
 309        RoundedRectangleBorder(
 310          borderRadius: BorderRadius.circular(AppRadius.small),
 311        ),
 312      );
 313      final label = WidgetStatePropertyAll(text.labelLarge);
 314      const tall = WidgetStatePropertyAll(Size(64, 48));
 315  
 316      // Pressed feedback stronger than Material's 10 % (08 §5, §8).
 317      WidgetStateProperty<Color?> overlay(Color c) =>
 318          WidgetStateProperty.resolveWith((states) {
 319            if (states.contains(WidgetState.pressed)) {
 320              return c.withValues(alpha: pressedOverlay);
 321            }
 322            if (states.contains(WidgetState.focused)) {
 323              return c.withValues(alpha: 0.12);
 324            }
 325            if (states.contains(WidgetState.hovered)) {
 326              return c.withValues(alpha: 0.08);
 327            }
 328            return null;
 329          });
 330      WidgetStateProperty<Color> onOff(Color on) =>
 331          WidgetStateProperty.resolveWith(
 332            (states) =>
 333                states.contains(WidgetState.disabled) ? p.textDisabled : on,
 334          );
 335      final secondary = ButtonStyle(
 336        minimumSize: tall,
 337        animationDuration: AppMotion.short,
 338        padding: const WidgetStatePropertyAll(
 339          EdgeInsets.symmetric(horizontal: 16, vertical: 8),
 340        ),
 341        shape: shape,
 342        textStyle: label,
 343        foregroundColor: onOff(p.textPrimary),
 344        iconColor: onOff(p.textPrimary),
 345        backgroundColor: WidgetStateProperty.resolveWith(
 346          (states) =>
 347              states.contains(WidgetState.pressed) ? hybrid.selection : null,
 348        ),
 349        // The selection fill is the pressed feedback; no extra overlay.
 350        overlayColor: WidgetStateProperty.resolveWith(
 351          (states) => states.contains(WidgetState.pressed)
 352              ? HybridTokens.transparent
 353              : overlay(p.textPrimary).resolve(states),
 354        ),
 355        side: WidgetStateProperty.resolveWith(
 356          (states) => BorderSide(
 357            color: states.contains(WidgetState.disabled)
 358                ? p.border
 359                : p.controlBorder,
 360          ),
 361        ),
 362      );
 363  
 364      // Primary action (styles.css, hybrid-refinement.css): the action fill
 365      // with its on-action text in every theme. WQA-01 (owner-accepted web
 366      // reference 86873ce): field mode's action is the bright red role with
 367      // black text (5.25:1), replacing the D-06 red outline; hover underlines
 368      // instead of darkening. Pressed: every Hybrid button takes the selection
 369      // fill with primary text (`button:active`). Disabled: opacity .55.
 370      // The label is `.primary-button`'s 16/600 (QA-04).
 371      final field = identical(hybrid, HybridTokens.field);
 372      Color disabled(Color c) => c.withValues(alpha: c.a * 0.55);
 373      return t.copyWith(
 374        filledButtonTheme: FilledButtonThemeData(
 375          style: ButtonStyle(
 376            minimumSize: tall,
 377            shape: shape,
 378            textStyle: WidgetStateProperty.resolveWith(
 379              (states) => text.bodyLarge?.copyWith(
 380                fontWeight: FontWeight.w600,
 381                height: 1.5,
 382                decoration: field && states.contains(WidgetState.hovered)
 383                    ? TextDecoration.underline
 384                    : TextDecoration.none,
 385              ),
 386            ),
 387            padding: const WidgetStatePropertyAll(
 388              EdgeInsets.symmetric(horizontal: 16, vertical: 8),
 389            ),
 390            animationDuration: AppMotion.short,
 391            backgroundColor: WidgetStateProperty.resolveWith((states) {
 392              if (states.contains(WidgetState.disabled)) {
 393                return disabled(hybrid.action);
 394              }
 395              if (states.contains(WidgetState.pressed)) return hybrid.selection;
 396              if (!field && states.contains(WidgetState.hovered)) {
 397                return Color.lerp(hybrid.action, const Color(0xFF000000), 0.13);
 398              }
 399              return hybrid.action;
 400            }),
 401            foregroundColor: WidgetStateProperty.resolveWith((states) {
 402              if (states.contains(WidgetState.disabled)) {
 403                return disabled(hybrid.onAction);
 404              }
 405              if (states.contains(WidgetState.pressed)) return p.textPrimary;
 406              return hybrid.onAction;
 407            }),
 408            iconColor: WidgetStateProperty.resolveWith((states) {
 409              if (states.contains(WidgetState.disabled)) {
 410                return disabled(hybrid.onAction);
 411              }
 412              if (states.contains(WidgetState.pressed)) return p.textPrimary;
 413              return hybrid.onAction;
 414            }),
 415            overlayColor: const WidgetStatePropertyAll(HybridTokens.transparent),
 416          ),
 417        ),
 418        outlinedButtonTheme: OutlinedButtonThemeData(style: secondary),
 419        // The flat design has no elevation: an elevated button is the
 420        // secondary role, on the surface.
 421        elevatedButtonTheme: ElevatedButtonThemeData(
 422          style: secondary.copyWith(
 423            backgroundColor: WidgetStateProperty.resolveWith(
 424              (states) => states.contains(WidgetState.disabled)
 425                  ? t.scaffoldBackgroundColor
 426                  : states.contains(WidgetState.pressed)
 427                  ? hybrid.selection
 428                  : s.surface,
 429            ),
 430            elevation: const WidgetStatePropertyAll(0),
 431          ),
 432        ),
 433        // plan.css `.quiet-button`: 14/500 secondary text, padding 8 12,
 434        // at least 48 × 48.
 435        textButtonTheme: TextButtonThemeData(
 436          style: ButtonStyle(
 437            minimumSize: const WidgetStatePropertyAll(Size(48, 48)),
 438            padding: const WidgetStatePropertyAll(
 439              EdgeInsets.symmetric(horizontal: 12, vertical: 8),
 440            ),
 441            animationDuration: AppMotion.short,
 442            shape: shape,
 443            textStyle: label,
 444            foregroundColor: onOff(p.textSecondary),
 445            iconColor: onOff(p.textSecondary),
 446            overlayColor: overlay(p.textPrimary),
 447          ),
 448        ),
 449        iconButtonTheme: IconButtonThemeData(
 450          style: ButtonStyle(
 451            minimumSize: const WidgetStatePropertyAll(Size(48, 48)),
 452            overlayColor: overlay(p.textPrimary),
 453          ),
 454        ),
 455        segmentedButtonTheme: SegmentedButtonThemeData(
 456          style: ButtonStyle(
 457            shape: shape,
 458            minimumSize: const WidgetStatePropertyAll(Size(48, 48)),
 459            textStyle: label,
 460            side: WidgetStatePropertyAll(BorderSide(color: p.controlBorder)),
 461            foregroundColor: onOff(p.textPrimary),
 462            backgroundColor: WidgetStateProperty.resolveWith(
 463              (states) => states.contains(WidgetState.selected)
 464                  ? hybrid.selection
 465                  : t.scaffoldBackgroundColor,
 466            ),
 467          ),
 468        ),

 472        // plan.css `.block-input input`: a 1 px border and 12 padding round

 477          constraints: const BoxConstraints(minHeight: 48),
 478          border: OutlineInputBorder(
 479            borderRadius: BorderRadius.circular(AppRadius.small),
 480          ),
 481          enabledBorder: OutlineInputBorder(
