void main(){
 testWidgets('current component specimens', (tester)async{
 await tester.runAsync(_loadFonts);
 tester.view.devicePixelRatio=1;tester.view.physicalSize=const Size(412,12000);
 addTearDown(tester.view.reset);
 final out=Directory('build/components-20261028/renders')..createSync(recursive:true);
 final manifest=<Map<String,Object?>>[];
 for(final theme in GalleryTheme.values){
 final samples=[...galleryEntries().where((w)=>w is! SizedBox).map((w){
 // The legacy gallery reused a Fits sentence in every verdict specimen.
 // Keep these isolated examples semantically consistent with their state.
 if(w is Card && w.child is Padding && (w.child as Padding).child is StatusBlock){
 final old=(w.child as Padding).child as StatusBlock;
 final state=old.state;
 final measured=state==FitState.fits||state==FitState.tight||state==FitState.doesNotFit;
 final reason=switch(state){FitState.fits=>'Time fits this illustrative window.',FitState.tight=>'Little time remains in this illustrative window.',FitState.doesNotFit=>'The planned time exceeds this illustrative window.',FitState.noWindow=>'No imaging window in this example.',FitState.needsInput=>'Choose a target before calculating fit.',FitState.nothingToFit=>'No light frames in this example.'};
 return Card(child:Padding(padding:const EdgeInsets.all(16),child:StatusBlock(state:state,needed:measured?Duration(minutes:state==FitState.doesNotFit?300:state==FitState.tight?250:125):null,usable:measured?const Duration(minutes:260):null,reason:reason)));
 }
 return w;
 }),...extras()];
 for(final (i,w) in samples.indexed){
 await tester.pumpWidget(const SizedBox());
 final data=galleryThemeData(theme);
 Widget content=ColoredBox(color:data.scaffoldBackgroundColor,child:Padding(padding:const EdgeInsets.all(16),child:Column(mainAxisSize:MainAxisSize.min,crossAxisAlignment:CrossAxisAlignment.stretch,children:[w])));
 if(theme==GalleryTheme.field)content=ColorFiltered(colorFilter:AppTheme.fieldFilter,child:content);
 await tester.pumpWidget(_app(theme,Scaffold(body:SingleChildScrollView(child:RepaintBoundary(key:_shot,child:content)))));
 await tester.pumpAndSettle();
 final error=tester.takeException();
 final types=find.byWidgetPredicate((_)=>true).evaluate().map((e)=>e.widget.runtimeType.toString()).toSet().toList();
 final name='component_${i.toString().padLeft(3,'0')}_${theme.name}';
 await _write(tester,'${out.path}/$name.png');
 manifest.add({'id':i,'theme':theme.name,'label':w.runtimeType.toString(),'file':'$name.png','types':types,'texts':find.byType(Text).evaluate().map((e)=>(e.widget as Text).data).whereType<String>().take(20).toList(),'error':error?.toString()});
 }
 }
 for(final theme in GalleryTheme.values){
 for(final kind in ['detail','confirm','unsaved','undo']){
 final name='${kind}_${theme.name}';
 await _screen(tester,theme,'${out.path}/$name.png',overlay:kind=='confirm'?(c){confirmDestructive(c,title:'Delete rig?',message:'Saved plans keep their values.');}:kind=='unsaved'?(c){askUnsavedChanges(c,plan:'M31');}:kind=='undo'?(c){showUndo(c,message:'Block removed',onUndo:(){});}:null);
 manifest.add({'id':kind,'theme':theme.name,'label':kind,'file':'$name.png','types':find.byWidgetPredicate((_)=>true).evaluate().map((e)=>e.widget.runtimeType.toString()).toSet().toList(),'texts':find.byType(Text).evaluate().map((e)=>(e.widget as Text).data).whereType<String>().take(20).toList(),'error':tester.takeException()?.toString()});
 }
 }
 File('${out.path}/manifest.json').writeAsStringSync(const JsonEncoder.withIndent('  ').convert(manifest));
 expect(manifest.where((m)=>m['error']!=null),isEmpty);
 });
