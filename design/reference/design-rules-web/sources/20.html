3 /// The motion scale (S5.2): short, subtle transitions only
4 /// (`.agents/rules/05-ui-design.md`). Every animated component asks
5 /// [duration] for its length, so the platform's reduced-motion setting
6 /// turns motion off. See `docs/DESIGN_SYSTEM.md`.
7 abstract final class AppMotion {
8 /// A state change in place: a pressed control, a chevron turning.
9 static const Duration short = Duration(milliseconds: 120);
10
11 /// Content appearing or collapsing: a section opening, a row leaving.
12 static const Duration medium = Duration(milliseconds: 180);
13
14 /// A brief mark on what an edit changed, fading out (S6.9: a capture
15 /// block just added or edited). Long enough to be seen, never repeated.
16 static const Duration highlight = Duration(milliseconds: 1500);
17
18 /// The one curve: fast out, gentle in.
19 static const Curve curve = Cubic(0.16, 1, 0.3, 1);
20
21 /// [base], or zero when the user asked the platform for less motion.
22 static Duration duration(BuildContext context, Duration base) =>
23 MediaQuery.maybeDisableAnimationsOf(context) ?? false
24 ? Duration.zero
25 : base;
26 }
