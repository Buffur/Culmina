243 /// Red field mode's safety net (owner decision, TASK 12.4): applied over
244 /// the whole app, it maps every pixel to red by brightness — R' = R +
245 /// 0.7152 G + 0.0722 B (Rec. 709 luma weights for G and B), G' = B' = 0 —
246 /// so map tiles and anything a token misses stay red. Pure red is
247 /// unchanged.
248 static const ColorFilter fieldFilter = ColorFilter.matrix(fieldFilterMatrix);
249
250 /// The 4×5 RGBA matrix of [fieldFilter], row by row.
251 static const List<double> fieldFilterMatrix = <double>[
252 1, 0.7152, 0.0722, 0, 0, //
253 0, 0, 0, 0, 0, //
254 0, 0, 0, 0, 0, //
255 0, 0, 0, 1, 0, //
256 ];

258 /// Adds the palette and the shared type scale (no text under 12 sp).
259 /// The scale's sizes and weights override Material's geometry; the
260 /// colours stay the theme's.
261 static ThemeData _withTokens(
262 ThemeData base,
263 AppPalette palette, {
264 Color? textColor,
265 bool field = false,
266 }) {
267 final hybrid = field
268 ? HybridTokens.field
269 : base.brightness == Brightness.dark
270 ? HybridTokens.dark
271 : HybridTokens.light;
272 TextTheme readable(TextTheme t) {
273 final scaled = t.merge(AppTypography.scale);
274 return textColor == null
275 ? scaled
276 : scaled.apply(bodyColor: textColor, displayColor: textColor);
277 }
278
279 return _withControls(
280 base.copyWith(
281 extensions: [
282 palette,
283 hybrid,
284 AppTextTheme.standard,
285 field
286 ? CulminaMarkColors.field
287 : base.brightness == Brightness.dark
288 ? CulminaMarkColors.dark
289 : CulminaMarkColors.light,
290 ],
291 pageTransitionsTheme: const PageTransitionsTheme(
292 builders: {
293 TargetPlatform.android: HybridPageTransitionsBuilder(),
294 TargetPlatform.windows: HybridPageTransitionsBuilder(),
295 TargetPlatform.linux: HybridPageTransitionsBuilder(),
296 TargetPlatform.iOS: CupertinoPageTransitionsBuilder(),
297 TargetPlatform.macOS: CupertinoPageTransitionsBuilder(),
298 },
299 ),
300 textTheme: readable(base.textTheme),
301 primaryTextTheme: readable(base.primaryTextTheme),
302 ),
303 palette,
304 field: field,
305 );
306 }

327 // Pressed feedback stronger than Material's 10 % (08 §5, §8).
328 WidgetStateProperty<Color?> overlay(Color c) =>
329 WidgetStateProperty.resolveWith((states) {
330 if (states.contains(WidgetState.pressed)) {
331 return c.withValues(alpha: pressedOverlay);
332 }
333 if (states.contains(WidgetState.focused)) {
334 return c.withValues(alpha: 0.12);
335 }
336 if (states.contains(WidgetState.hovered)) {
337 return c.withValues(alpha: 0.08);
338 }
339 return null;
340 });
341 WidgetStateProperty<Color> onOff(Color on) =>
342 WidgetStateProperty.resolveWith(
343 (states) =>
344 states.contains(WidgetState.disabled) ? p.textDisabled : on,

370 : p.controlBorder,
371 ),
372 ),

400 ),

416 if (states.contains(WidgetState.pressed)) return p.textPrimary;
417 return hybrid.onAction;
418 }),
419 iconColor: WidgetStateProperty.resolveWith((states) {
420 if (states.contains(WidgetState.disabled)) {
421 return disabled(hybrid.onAction);
422 }
423 if (states.contains(WidgetState.pressed)) return p.textPrimary;

425 }),
426 overlayColor: const WidgetStatePropertyAll(HybridTokens.transparent),
427 ),
428 ),
429 outlinedButtonTheme: OutlinedButtonThemeData(style: secondary),

431 // secondary role, on the surface.

433 style: secondary.copyWith(
434 backgroundColor: WidgetStateProperty.resolveWith(
435 (states) => states.contains(WidgetState.disabled)

437 : states.contains(WidgetState.pressed)
438 ? hybrid.selection
439 : s.surface,
