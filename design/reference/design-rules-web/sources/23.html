12 /// A plan's state as the user sees it (S5.4; ADR-019 §3, RD-05, RG-04).
13 /// The stored statuses stay internal: "Draft" is never shown.
14 enum PlanState {
15 notSaved,
16 saved,
17 savedChanged,
18 tracking,
19 completed,
20
21 /// A partial result, recorded through the result form (S8.1–S8.2).
22 partly,
23 notDone,
24 oldLog;
25
26 /// The state the stored fields express: the status, whether the plan was
27 /// ever saved (`plannedAtUtc != null`), whether it is a legacy log, and how
28 /// a completed result was reported (S8.1). Pure.
29 static PlanState from({
30 required SessionStatus status,
31 required bool savedBefore,
32 required bool legacy,
33 ResultKind? resultKind,
34 }) {
35 if (legacy) return oldLog;
36 return switch (status) {
37 SessionStatus.draft => savedBefore ? savedChanged : notSaved,
38 SessionStatus.planned => saved,
39 SessionStatus.inProgress => tracking,
40 SessionStatus.completed =>
41 resultKind == ResultKind.partly ? partly : completed,
42 SessionStatus.abandoned => notDone,
43 };
44 }
45
46 static PlanState of(Session s) => from(
47 status: s.status,
48 savedBefore: s.plannedAtUtc != null,
49 legacy: s.legacy,
50 resultKind: s.resultKind,
51 );
52
53 /// Current planner facts can be newer than the stored row while autosave
54 /// is pending or failed. Only explicit Save plan settles those edits.
55 static PlanState forWorkingPlan(
56 Session s, {
57 required bool hasUnsavedChanges,
58 }) {
59 final stored = of(s);
60 return stored == saved && hasUnsavedChanges ? savedChanged : stored;
61 }

63 String get word => switch (this) {
64 notSaved => AppWords.notSaved,
65 saved => AppWords.saved,
66 savedChanged => AppWords.savedChanged,
67 tracking => AppWords.tracking,
68 completed => AppWords.completed,
69 partly => AppWords.partly,
70 notDone => AppWords.notDone,
71 oldLog => AppWords.oldLog,
72 };

74 /// F19 A1: a result's state carries an icon beside its word (never
75 /// colour alone); the others are words only.
76 IconData? get icon => switch (this) {
77 completed => Icons.check_circle_outline,
78 partly => Icons.incomplete_circle_outlined,
79 notDone => Icons.cancel_outlined,
80 _ => null,
81 };
82
83 /// Unsaved work draws attention; a settled plan reads as primary text;
84 /// the rest is quiet. A tone, never a verdict.
85 Color tone(AppPalette p) => switch (this) {
86 notSaved || savedChanged => p.stateUnsaved,
87 saved || tracking || completed => p.stateSettled,
88 partly || notDone || oldLog => p.stateQuiet,
89 };
90 }

92 /// The plan-state label (S5.4) as the Hybrid reference's compact status
93 /// (notion-hybrid.css): Saved, Tracking and Completed in the success role on
94 /// its wash; Not saved, Saved · changed and Partly in the caution role on
95 /// its wash (attention, not an error); Not done and Old log quiet on the
96 /// selection fill. 13 px, padding 3 × 6, radius 4. [withIcon]: Tonight's
97 /// check (saved) or pencil (unsaved work) mark, 13 px; Plan's label is
98 /// the word alone at weight 500. It wraps at large text; not tappable.
99 class PlanStateLabel extends StatelessWidget {
100 const PlanStateLabel(this.state, {super.key, this.withIcon = false});
101
102 final PlanState state;
103 final bool withIcon;
104
105 @override
106 Widget build(BuildContext context) {
107 final p = AppPalette.of(context);
108 final h = Theme.of(context).extension<HybridTokens>() ?? HybridTokens.light;
109 final (fg, bg) = switch (state) {
110 PlanState.saved ||
111 PlanState.tracking ||
112 PlanState.completed => (h.success, h.successWash),
113 PlanState.notSaved ||
114 PlanState.savedChanged ||
115 PlanState.partly => (h.caution, h.cautionWash),
116 PlanState.notDone || PlanState.oldLog => (p.textPrimary, h.selection),

119 state == PlanState.saved ||
120 state == PlanState.tracking ||
