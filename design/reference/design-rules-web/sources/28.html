12 /// The bottom-navigation shell of ADR-015 (TASK 12.2): Tonight · Logbook ·
13 /// Library · Settings, one navigator per tab.
14 ///
15 /// Back (ADR-015 §3): pops within the current tab first (go_router); at a
16 /// tab's root it returns to Tonight; at Tonight's root it leaves the app.
17 class AppShell extends StatelessWidget {
18 const AppShell({
19 super.key,
20 required this.navigationShell,
21 this.showNavigation = true,
22 });
23
24 final StatefulNavigationShell navigationShell;
25 final bool showNavigation;
26
27 /// The root navigation bar (for tests and the host renders).
28 static const navigationKey = Key('app.navigation');
29
30 /// The four roots with the reference's line icons (workspace.js `paths`).
31 static List<({HybridIconData icon, String label})> get items => [
32 (icon: HybridIcons.moon, label: 'Tonight'),
33 if (FeatureScope.logbook)
34 (icon: HybridIcons.book, label: AppWords.logbook), // S8.5 (RD-14)
35 (icon: HybridIcons.library, label: 'Library'),
36 (icon: HybridIcons.settings, label: 'Settings'),
37 ];
38
39 /// Kept for callers that list the destinations' labels.
40 static List<NavigationDestination> get destinations => [
41 for (final i in items)
42 NavigationDestination(icon: HybridIcon(i.icon), label: i.label),
43 ];

45 @override
46 Widget build(BuildContext context) {
47 final onTonight = navigationShell.currentIndex == 0;
48 void select(int i) => navigationShell.goBranch(
49 i,
50 initialLocation: i == navigationShell.currentIndex,
51 );
52 final largeText = MediaQuery.textScalerOf(context).scale(1) >= 1.5;
53 final theme = Theme.of(context);
54 final palette = AppPalette.of(context);
55 final hybrid = theme.extension<HybridTokens>() ?? HybridTokens.light;
56 // workspace.css `.app-nav` (Hybrid): a subtle top rule, padding 8 4,
57 // four equal 56 px buttons 2 apart; the selected one on the subtle
58 // surface (radius 6) in 600 primary, the others 400 secondary; a 19 px
59 // icon 5 above the 11 px Geist label. At large text, two per row.
60 Widget button(int i) {
61 final selected = i == navigationShell.currentIndex;
62 final color = selected ? palette.textPrimary : palette.textSecondary;
63 final item = items[i];
64 return Semantics(
65 selected: selected,
66 button: true,
67 label: item.label,
68 excludeSemantics: true,
69 enabled: true,
70 onTap: () => select(i),
71 child: Material(
72 color: selected ? hybrid.surfaceSubtle : HybridTokens.transparent,
73 borderRadius: BorderRadius.circular(6),
74 child: InkWell(
75 key: Key('nav.${item.label.toLowerCase()}'),
76 borderRadius: BorderRadius.circular(6),
77 onTap: () => select(i),
78 child: ConstrainedBox(
79 constraints: const BoxConstraints(minHeight: 56),
80 child: Padding(
81 padding: const EdgeInsets.all(4),
82 child: Center(
83 child: Wrap(
84 direction: largeText ? Axis.horizontal : Axis.vertical,
85 alignment: WrapAlignment.center,
86 crossAxisAlignment: WrapCrossAlignment.center,
87 spacing: largeText ? 8 : 5,
88 children: [
89 HybridIcon(
90 item.icon,
91 size: MediaQuery.textScalerOf(context).scale(19),
92 color: color,
93 ),
94 Text(
95 item.label,
96 textAlign: TextAlign.center,
97 style: AppTextTheme.of(context).navigationLabel
98 .copyWith(
99 color: color,
100 fontWeight: selected
101 ? FontWeight.w600
102 : FontWeight.w400,
103 ),
104 ),
105 ],
106 ),
107 ),
108 ),
109 ),
110 ),
111 ),
112 );
113 }
114
115 final perRow = largeText ? 2 : items.length;
116 return PopScope(
117 canPop: onTonight,
118 onPopInvokedWithResult: (didPop, _) {
119 if (!didPop) navigationShell.goBranch(0);
120 },
121 child: Scaffold(
122 body: HybridViewEntrance(
123 trigger: navigationShell.currentIndex,
124 child: navigationShell,
125 ),
126 bottomNavigationBar: !showNavigation
127 ? null
128 : DecoratedBox(
129 key: navigationKey,
130 decoration: BoxDecoration(
131 color: theme.scaffoldBackgroundColor,
132 border: Border(top: BorderSide(color: palette.border)),
133 ),
134 child: SafeArea(
135 top: false,
136 child: Padding(
137 padding: const EdgeInsets.symmetric(
138 horizontal: 4,
139 vertical: 8,
140 ),
141 child: Column(
142 mainAxisSize: MainAxisSize.min,
143 children: [
144 for (var row = 0; row < items.length; row += perRow)
145 Padding(
146 padding: EdgeInsets.only(top: row == 0 ? 0 : 2),
147 child: Row(
148 children: [
149 for (
150 var i = row;
151 i < row + perRow && i < items.length;
152 i++
153 ) ...[
154 if (i > row) const SizedBox(width: 2),
155 Expanded(child: button(i)),
156 ],
157 ],
158 ),
159 ),
160 ],
161 ),
162 ),
163 ),
164 ),
165 ),
166 );
167 }
