12 /// Detail one tap away (S5.5; ADR-019 §7, RD-06, RG-06): a header with the
13 /// section's title and a factual summary, and the content below it when
14 /// open. The open or closed state is remembered per [sectionKey]
15 /// ([DisclosureViewModel]).
16 ///
17 /// The [summary] states facts ("3 lines · 2 h 35 min total"), never a
18 /// verdict ("good", "fine"); the verdict belongs to the status block. The
19 /// summary stays visible when the section is open.
20 class CollapsibleSection extends StatelessWidget {
21 const CollapsibleSection({
22 super.key,
23 required this.sectionKey,
24 required this.title,
25 required this.child,
26 this.summary,
27 this.initiallyOpen = false,
28 this.open,
29 this.onToggle,
30 this.topRule = true,
31 this.titleStyle,
32 this.icon,
33 this.iconColor,
34 this.readableFeedback = false,
35 });
36
37 /// A data icon before the title (conditions.css `.condition-data-heading`:
38 /// 20 px, 10 before the words), e.g. the Moon or the Sun.
39 final HybridIconData? icon;
40 final Color? iconColor;
41
42 /// Local opt-in for confirmed quiet-label interaction states.
43 final bool readableFeedback;
44
45 /// The Hybrid disclosure's thin rule above it (`.plan-disclosure`).
46 final bool topRule;
47
48 /// A nested disclosure's smaller title (Library 15/600); defaults to the
49 /// section heading role 20/600.
50 final TextStyle? titleStyle;
51
52 /// Stable and unique, e.g. `planner.budgetDetails`; it names the stored
53 /// preference, so never rename one without a reason.
54 final String sectionKey;
55 final String title;
56 final String? summary;
57 final Widget child;
58
59 /// The state before the user has opened or closed it.
60 final bool initiallyOpen;
61
62 /// Controlled use (S7.6): the caller holds the state, e.g. a dialog whose
63 /// section need not be remembered. With both set, the section reads no
64 /// [DisclosureViewModel] and [initiallyOpen] is unused.
65 final bool? open;
66 final VoidCallback? onToggle;
67
68 @override
69 Widget build(BuildContext context) {
70 final controlled = this.open != null && onToggle != null;
71 final vm = controlled ? null : context.watch<DisclosureViewModel>();
72 final open =
73 this.open ?? vm!.isOpen(sectionKey, initiallyOpen: initiallyOpen);
74 final p = AppPalette.of(context);
75 void toggle() => controlled ? onToggle!() : vm!.setOpen(sectionKey, !open);
76 final resize = AppMotion.duration(context, AppMotion.medium);
77 final Widget body = open
78 ? Padding(
79 padding: const EdgeInsets.only(bottom: AppSpacing.sm),
80 child: child,
81 )
82 : const SizedBox(width: double.infinity);
83
84 // The Hybrid disclosure (hybrid-refinement.css `.plan-disclosure`):
85 // grid title | chevron, gap 16, min 48, padding 6 0; title 20/600 at
86 // 1.4; caption 13 at 1.4, 2 below; an 18 px chevron (m9 5 7 7-7 7,
87 // stroke 1.7) turned down when closed and up when open (120 ms).
88 final header = Semantics(
89 key: Key('section.$sectionKey'),
90 button: true,
91 enabled: true,
92 expanded: open,
93 onTap: toggle,
94 hint: open ? 'Collapse' : 'Expand',
95 excludeSemantics: true,
96 label: [title, ?summary].join(', '),
97 child: InkWell(
98 overlayColor: readableFeedback ? HybridQuietInk.overlay(context) : null,
99 onTap: toggle,
100 child: ConstrainedBox(
101 constraints: const BoxConstraints(minHeight: 48),
102 child: Padding(
103 padding: const EdgeInsets.symmetric(vertical: 6),
104 child: Row(
105 children: [
106 Expanded(
107 child: Column(
108 crossAxisAlignment: CrossAxisAlignment.start,
109 children: [
110 Row(
111 children: [
112 if (icon case final i?) ...[
113 HybridIcon(
114 i,
115 size: 20,
116 strokeWidth: 1.7,
117 color: iconColor ?? p.textPrimary,
118 ),
119 const SizedBox(width: 10),
120 ],
121 Flexible(
122 child: Text(
123 title,
124 style:
125 (titleStyle ??
126 AppTextTheme.of(context).resolve(
127 AppTextTheme.of(context)
128 .sectionHeading,
129 AppTextGeometry.badge,
130 ))
131 .copyWith(color: p.textPrimary),
132 ),
133 ),
134 ],
135 ),
136 if (summary != null)
137 Padding(
138 padding: const EdgeInsets.only(top: 2),
139 child: Text(
140 summary!,
141 style: AppTextTheme.of(context)
142 .resolve(
143 AppTextTheme.of(context).caption,
144 AppTextGeometry.compact,
145 )
146 .copyWith(color: p.textSecondary),
147 ),
148 ),
149 ],
150 ),
151 ),
152 const SizedBox(width: AppSpacing.md),
153 AnimatedRotation(
154 turns: open ? -0.25 : 0.25,
155 duration: AppMotion.duration(context, AppMotion.short),
156 curve: AppMotion.curve,
157 child: HybridIcon(
158 HybridIcons.chevron,
159 size: 18,
160 color: p.textSecondary,
161 ),
162 ),
163 ],
164 ),
165 ),
166 ),
167 ),
168 );
169
170 return Column(
171 crossAxisAlignment: CrossAxisAlignment.stretch,
172 children: [
173 if (topRule)
174 DecoratedBox(
175 decoration: BoxDecoration(
176 border: Border(top: BorderSide(color: p.border)),
177 ),
178 child: header,
179 )
180 else
181 header,
182 // Reduced motion: no AnimatedSize at all (with a zero duration it
183 // throws a layout assertion).
184 if (resize == Duration.zero)
185 body
186 else
187 ClipRect(
188 child: AnimatedSize(
189 duration: resize,
190 curve: AppMotion.curve,
191 alignment: Alignment.topCenter,
192 child: body,
193 ),
194 ),
195 ],
196 );
197 }
198 }
