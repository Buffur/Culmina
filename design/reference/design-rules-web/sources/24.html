12 /// The answer first (S5.4; ADR-019 §6, addendum §3.1–§3.2): the verdict
13 /// headline in its status colour ("Fits: 2 h 5 min needed of 4 h 20 min
14 /// usable"), its reason, optional key numbers and an optional action (fill
15 /// or trim). It calculates nothing: the state, the durations, the reason
16 /// and the key numbers come from the ViewModel (the fit analysis); values
17 /// are formatted with `QuantityText`.
18 class StatusBlock extends StatelessWidget {
19 const StatusBlock({
20 super.key,
21 required this.state,
22 this.needed,
23 this.usable,
24 this.reason,
25 this.missing,
26 this.keyNumbers = const [],
27 this.action,
28 this.separateFigures = false,
29 this.headingStyle,
30 this.figuresBeforeReason,
31 this.fitLine = false,
32 });

34 /// Tonight's Hybrid fit line (app.js `.fit-line`, `.plan-reason`): a
35 /// check (or an info mark for a missing input) beside a 13 px verdict, the
36 /// reason in 14 px secondary below it.
37 final bool fitLine;
38
39 final FitState state;
40
41 /// Time needed and usable time; the headline names them only when both
42 /// are known (unknown stays unknown).
43 final Duration? needed;
44 final Duration? usable;
45
46 final String? reason;
47
48 /// For a missing input: [AppWords.needsTarget] (the default) or another
49 /// glossary word the adopting screen chooses.
50 final String? missing;
51
52 /// Label and already formatted value pairs, e.g. ("Capture ends",
53 /// "01:40").
54 final List<(String, String)> keyNumbers;
55
56 final Widget? action;
57
58 /// Hybrid shows the verdict, then labelled measurements rather than a
59 /// sentence that repeats both values in the heading.
60 final bool separateFigures;
61 final TextStyle? headingStyle;
62
63 /// Plan shows figures before its explanation; Tonight retains its summary
64 /// reason above them. Existing sentence-headline adopters keep their order.
65 final bool? figuresBeforeReason;
66
67 /// The headline in the glossary's words (RD-14). Pure.
68 static String headline(
69 FitState state, {
70 Duration? needed,
71 Duration? usable,
72 String? missing,
73 }) {
74 String measured(String word) => needed == null || usable == null
75 ? word
76 : AppWords.headline(
77 word,
78 QuantityText.budgetDuration(needed),
79 QuantityText.budgetDuration(usable),
80 );
81 return switch (state) {
82 FitState.fits => measured(AppWords.fits),
83 FitState.tight => measured(AppWords.tight),
84 FitState.doesNotFit => measured(AppWords.doesNotFit),
85 FitState.noWindow => AppWords.noWindow,
86 FitState.needsInput => missing ?? AppWords.needsTarget,
87 FitState.nothingToFit => AppWords.needsBlock,
88 };
89 }

91 @override
92 Widget build(BuildContext context) {
93 final theme = Theme.of(context);
94 final p = AppPalette.of(context);
95 final text = theme.textTheme;
96 final figures = [
97 if (separateFigures && needed != null)
98 ('Needed in imaging window', QuantityText.budgetDuration(needed!)),
99 if (separateFigures && usable != null)
100 ('Usable', QuantityText.budgetDuration(usable!)),
101 ...keyNumbers,
102 ];
103 final figuresFirst = figuresBeforeReason ?? separateFigures;
104 final explanation = Text(
105 reason ?? '',
106 style: fitLine
107 ? AppTextTheme.of(context)
108 .resolve(text.bodySmall, AppTextGeometry.fitExplanation)
109 .copyWith(color: p.textSecondary)
110 : text.bodyMedium?.copyWith(color: p.textSecondary),
111 );
112 final verdictStyle =
113 (fitLine
114 ? AppTextTheme.of(context)
115 .resolve(text.bodySmall, AppTextGeometry.fitVerdict)
116 .copyWith(fontWeight: FontWeight.w500)
117 : (headingStyle ?? text.titleMedium))
118 ?.copyWith(color: FitText.color(state, theme.colorScheme, p));
119 final verdict = Text(
120 headline(
121 state,
122 needed: separateFigures ? null : needed,
123 usable: separateFigures ? null : usable,
124 missing: missing,
125 ),
126 key: const Key('status.headline'),
127 style: verdictStyle,
128 );
129 return Column(
130 crossAxisAlignment: CrossAxisAlignment.start,
131 children: [
132 Semantics(
133 header: true,
134 child: fitLine
135 ? Row(
136 crossAxisAlignment: CrossAxisAlignment.start,
137 children: [
138 Padding(
139 padding: const EdgeInsets.only(top: 1, right: 7),
140 child: Icon(
141 state == FitState.needsInput ||
142 state == FitState.nothingToFit
143 ? Icons.info_outline
144 : state == FitState.fits || state == FitState.tight
145 ? Icons.check
146 : Icons.close,
147 // 17 px at 100 %, growing with the text it marks.
148 size: MediaQuery.textScalerOf(context).scale(17),
149 color: verdictStyle?.color,
150 ),
151 ),
152 Expanded(child: verdict),
153 ],
154 )
155 : verdict,
156 ),
157 if (reason != null && !figuresFirst) ...[
158 // Hybrid fit line: the reason 6 below the verdict.
159 SizedBox(height: fitLine ? 6 : AppSpacing.xs),
160 explanation,
161 ],
162 if (figures.isNotEmpty) ...[
163 const SizedBox(height: AppSpacing.md),
164 // S6.16: each number as a label above its value, so the answer's
165 // figures read as a set (the scale's caption and group roles).
166 Wrap(
167 spacing: AppSpacing.lg,
168 runSpacing: AppSpacing.sm,
169 children: [
170 for (final (label, value) in figures)
171 MergeSemantics(
172 child: Column(
173 key: Key('status.number.$label'),
174 crossAxisAlignment: CrossAxisAlignment.start,
175 mainAxisSize: MainAxisSize.min,
176 children: [
177 Text(
178 label,
179 // Hybrid `.plan-measures dt`: 13 at normal leading.
180 style:
181 (fitLine
182 ? AppTextTheme.of(context).resolve(
183 text.bodySmall,
184 AppTextGeometry.fitMeasure,
185 )
186 : text.bodySmall)
187 ?.copyWith(color: p.textTertiary),
188 ),
189 if (fitLine) const SizedBox(height: AppSpacing.xs),
190 Text(
191 value,
192 style: fitLine
193 ? AppTextTheme.of(context)
194 .resolve(
195 text.bodyMedium,
196 AppTextGeometry.fitValue,
197 )
198 .copyWith(
199 color: p.textPrimary,
200 fontWeight: FontWeight.w500,
201 fontFeatures: const [
202 FontFeature.tabularFigures(),
203 ],
204 )
205 : text.titleSmall?.copyWith(color: p.textPrimary),
206 ),
207 ],
208 ),
209 ),
210 ],
211 ),
212 ],
213 if (reason != null && figuresFirst) ...[
214 const SizedBox(height: AppSpacing.sm),
215 explanation,
216 ],
217 if (action != null) ...[const SizedBox(height: AppSpacing.md), action!],
218 ],
219 );
220 }
