72 /// The boundary of a control (S5.2): a text field's underline or

424 @override
425 AppPalette lerp(AppPalette? other, double t) {
426 if (other == null || t == 0) return this;
427 if (t == 1) return other;
428 assert(bortle.length == other.bortle.length);
429 assert(onBortle.length == other.onBortle.length);
430 return AppPalette(
431 fieldMode: t < .5 ? fieldMode : other.fieldMode,
432 brightness: t < .5 ? brightness : other.brightness,
433 textPrimary: Color.lerp(textPrimary, other.textPrimary, t)!,
434 textSecondary: Color.lerp(textSecondary, other.textSecondary, t)!,
435 textTertiary: Color.lerp(textTertiary, other.textTertiary, t)!,
436 textDisabled: Color.lerp(textDisabled, other.textDisabled, t)!,
437 surfaceRaised: Color.lerp(surfaceRaised, other.surfaceRaised, t)!,
438 border: Color.lerp(border, other.border, t)!,
439 controlBorder: Color.lerp(controlBorder, other.controlBorder, t)!,
440 statusFits: Color.lerp(statusFits, other.statusFits, t)!,
441 statusTight: Color.lerp(statusTight, other.statusTight, t)!,
442 statusDoesNotFit: Color.lerp(
443 statusDoesNotFit,
444 other.statusDoesNotFit,
445 t,
446 )!,
447 statusNoWindow: Color.lerp(statusNoWindow, other.statusNoWindow, t)!,
448 statusNeutral: Color.lerp(statusNeutral, other.statusNeutral, t)!,
449 stateUnsaved: Color.lerp(stateUnsaved, other.stateUnsaved, t)!,
450 stateSettled: Color.lerp(stateSettled, other.stateSettled, t)!,
451 stateQuiet: Color.lerp(stateQuiet, other.stateQuiet, t)!,
452 muted: Color.lerp(muted, other.muted, t)!,
453 caution: Color.lerp(caution, other.caution, t)!,
454 sunEvent: Color.lerp(sunEvent, other.sunEvent, t)!,
455 twilightEvent: Color.lerp(twilightEvent, other.twilightEvent, t)!,
456 moon: Color.lerp(moon, other.moon, t)!,
457 selected: Color.lerp(selected, other.selected, t)!,
458 chartDay: Color.lerp(chartDay, other.chartDay, t)!,
459 chartTwilight: Color.lerp(chartTwilight, other.chartTwilight, t)!,
460 chartDark: Color.lerp(chartDark, other.chartDark, t)!,
461 chartWindow: Color.lerp(chartWindow, other.chartWindow, t)!,
462 chartTarget: Color.lerp(chartTarget, other.chartTarget, t)!,
463 chartMoon: Color.lerp(chartMoon, other.chartMoon, t)!,
464 chartMinAltitude: Color.lerp(
465 chartMinAltitude,
466 other.chartMinAltitude,
467 t,
468 )!,
469 chartNow: Color.lerp(chartNow, other.chartNow, t)!,
470 chartNowCentre: Color.lerp(chartNowCentre, other.chartNowCentre, t)!,
471 chartGrid: Color.lerp(chartGrid, other.chartGrid, t)!,
472 chartLabel: Color.lerp(chartLabel, other.chartLabel, t)!,
473 swatchBorder: Color.lerp(swatchBorder, other.swatchBorder, t)!,
474 bortle: List.generate(
475 bortle.length,
476 (i) => Color.lerp(bortle[i], other.bortle[i], t)!,
477 ),
478 onBortle: List.generate(
479 onBortle.length,
