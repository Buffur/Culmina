150 /// Retains the caller's inherited family/color/weight and replaces only the
151 /// centrally declared geometry. A null Material role stays inherited.
152 TextStyle resolve(TextStyle? base, AppTextGeometry geometry) =>
153 (base ?? inherited).merge(_geometry[geometry]);

202 @override
203 AppTextTheme lerp(covariant AppTextTheme? other, double t) {
204 if (other == null) return this;
205 return AppTextTheme._(
206 {
207 for (final key in _roles.keys)
208 key: TextStyle.lerp(_roles[key], other._roles[key], t)!,
209 },
210 {
211 for (final key in _geometry.keys)
212 key: TextStyle.lerp(_geometry[key], other._geometry[key], t)!,
213 },
214 );
215 }
